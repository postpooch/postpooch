CREATE SCHEMA IF NOT EXISTS public;

CREATE TABLE public.blog_posts (
  id uuid DEFAULT gen_random_uuid() NOT NULL,
  created_at timestamp with time zone DEFAULT now() NOT NULL,
  text text,
  website_id uuid,
  title text,
  subtitle text,

CREATE TABLE public.crm_activities (
  id uuid DEFAULT gen_random_uuid() NOT NULL,
  profile_id uuid NOT NULL,
  contact_id uuid,
  property_id uuid,
  transaction_id uuid,
  task_id uuid,
  activity_type text NOT NULL,
  summary text NOT NULL,
  metadata jsonb DEFAULT '{}'::jsonb NOT NULL,
  created_at timestamp with time zone DEFAULT now() NOT NULL,
  CONSTRAINT crm_activities_activity_type_check CHECK (activity_type = ANY (ARRAY['contact_created'::text, 'contact_updated'::text, 'property_created'::text, 'property_updated'::text, 'transaction_created'::text, 'transaction_updated'::text, 'stage_changed'::text, 'task_created'::text, 'task_completed'::text, 'note_added'::text])),
  CONSTRAINT crm_activities_contact_id_profile_id_fkey FOREIGN KEY (contact_id, profile_id) REFERENCES crm_contacts(id, profile_id) ON UPDATE CASCADE ON DELETE CASCADE,
  CONSTRAINT crm_activities_id_profile_id_key UNIQUE (id, profile_id),
  CONSTRAINT crm_activities_metadata_check CHECK (jsonb_typeof(metadata) = 'object'::text),
  CONSTRAINT crm_activities_pkey PRIMARY KEY (id),
  CONSTRAINT crm_activities_profile_id_fkey FOREIGN KEY (profile_id) REFERENCES profiles(id) ON UPDATE CASCADE ON DELETE CASCADE,
  CONSTRAINT crm_activities_property_id_profile_id_fkey FOREIGN KEY (property_id, profile_id) REFERENCES crm_properties(id, profile_id) ON UPDATE CASCADE ON DELETE CASCADE,
  CONSTRAINT crm_activities_task_id_profile_id_fkey FOREIGN KEY (task_id, profile_id) REFERENCES crm_tasks(id, profile_id) ON UPDATE CASCADE ON DELETE CASCADE,
  CONSTRAINT crm_activities_transaction_id_profile_id_fkey FOREIGN KEY (transaction_id, profile_id) REFERENCES crm_transactions(id, profile_id) ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE public.crm_contacts (
  id uuid DEFAULT gen_random_uuid() NOT NULL,
  profile_id uuid NOT NULL,
  first_name text NOT NULL,
  last_name text DEFAULT ''::text NOT NULL,
  email text,
  phone text,
  contact_types text[] DEFAULT '{}'::text[] NOT NULL,
  lead_status text DEFAULT 'new'::text NOT NULL,
  lead_source text,
  priority text DEFAULT 'warm'::text NOT NULL,
  price_min numeric(14,2),
  price_max numeric(14,2),
  preferred_locations text[] DEFAULT '{}'::text[] NOT NULL,
  preapproved boolean DEFAULT false NOT NULL,
  last_contacted_at timestamp with time zone,
  next_follow_up_at timestamp with time zone,
  notes text,
  created_at timestamp with time zone DEFAULT now() NOT NULL,
  updated_at timestamp with time zone DEFAULT now() NOT NULL,
  stage_id uuid,
  email_marketing_status text DEFAULT 'unknown'::text NOT NULL,
  email_marketing_status_changed_at timestamp with time zone,
  CONSTRAINT crm_contacts_check CHECK (price_min IS NULL OR price_max IS NULL OR price_max >= price_min),
  CONSTRAINT crm_contacts_contact_types_check CHECK (cardinality(contact_types) <= 8),
  CONSTRAINT crm_contacts_email_marketing_status_check CHECK (email_marketing_status = ANY (ARRAY['unknown'::text, 'subscribed'::text, 'unsubscribed'::text, 'suppressed'::text, 'invalid'::text])) NOT VALID,
  CONSTRAINT crm_contacts_id_profile_id_key UNIQUE (id, profile_id),
  CONSTRAINT crm_contacts_lead_status_check CHECK (lead_status = ANY (ARRAY['new'::text, 'contacted'::text, 'qualified'::text, 'nurturing'::text, 'active_client'::text, 'past_client'::text, 'inactive'::text])),
  CONSTRAINT crm_contacts_pkey PRIMARY KEY (id),
  CONSTRAINT crm_contacts_preferred_locations_check CHECK (cardinality(preferred_locations) <= 20),
  CONSTRAINT crm_contacts_price_max_check CHECK (price_max IS NULL OR price_max >= 0::numeric),
  CONSTRAINT crm_contacts_price_min_check CHECK (price_min IS NULL OR price_min >= 0::numeric),
  CONSTRAINT crm_contacts_priority_check CHECK (priority = ANY (ARRAY['hot'::text, 'warm'::text, 'cold'::text])),
  CONSTRAINT crm_contacts_profile_id_fkey FOREIGN KEY (profile_id) REFERENCES profiles(id) ON UPDATE CASCADE ON DELETE CASCADE,
  CONSTRAINT crm_contacts_stage_id_fkey FOREIGN KEY (stage_id) REFERENCES crm_contact_stages(id) ON UPDATE CASCADE ON DELETE SET NULL
);

CREATE OR REPLACE FUNCTION public.acquire_posting_run(p_schedule_id uuid, p_occurrence_at timestamp with time zone, p_trigger_run_id text, p_zernio_request_id uuid, p_lease_seconds integer DEFAULT 300)
 RETURNS TABLE(id uuid, schedule_id uuid, occurrence_at timestamp with time zone, trigger_run_id text, status text, template_id uuid, zernio_request_id uuid, zernio_post_id text, media_urls jsonb, acquired boolean)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  v_run public.posting_runs%rowtype;
begin
  if p_lease_seconds < 30 or p_lease_seconds > 3600 then
    raise exception 'p_lease_seconds must be between 30 and 3600';
  end if;

  insert into public.posting_runs (
    schedule_id,
    occurrence_at,
    trigger_run_id,
    zernio_request_id
  )
  values (
    p_schedule_id,
    p_occurrence_at,
    p_trigger_run_id,
    p_zernio_request_id
  )
  on conflict on constraint posting_runs_schedule_occurrence_key do nothing;

  select pr.*
  into v_run
  from public.posting_runs as pr
  where pr.schedule_id = p_schedule_id
    and pr.occurrence_at = p_occurrence_at
  for update;

  if v_run.status = 'published' then
    return query
    select
      v_run.id,
      v_run.schedule_id,
      v_run.occurrence_at,
      v_run.trigger_run_id,
      v_run.status,
      v_run.template_id,
      v_run.zernio_request_id,
      v_run.zernio_post_id,
      v_run.media_urls,
      false;
    return;
  end if;

  if v_run.status = 'processing'
    and v_run.lease_expires_at > now()
    and v_run.trigger_run_id is distinct from p_trigger_run_id
  then
    return query
    select
      v_run.id,
      v_run.schedule_id,
      v_run.occurrence_at,
      v_run.trigger_run_id,
      v_run.status,
      v_run.template_id,
      v_run.zernio_request_id,
      v_run.zernio_post_id,
      v_run.media_urls,
      false;
    return;
  end if;

  update public.posting_runs as pr
  set
    trigger_run_id = p_trigger_run_id,
    status = 'processing',
    attempt_count = pr.attempt_count + 1,
    lease_expires_at = now() + make_interval(secs => p_lease_seconds),
    updated_at = now(),
    last_error = null
  where pr.id = v_run.id
  returning pr.* into v_run;

  return query
  select
    v_run.id,
    v_run.schedule_id,
    v_run.occurrence_at,
    v_run.trigger_run_id,
    v_run.status,
    v_run.template_id,
    v_run.zernio_request_id,
    v_run.zernio_post_id,
    v_run.media_urls,
    true;
end;
$function$


CREATE OR REPLACE FUNCTION public.finalize_context_user_post(p_posting_run_id uuid)
 RETURNS boolean
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  v_profile_id uuid;
  v_status text;
  v_published_at timestamptz;
  v_history_summary text;
  v_visual_qa_status text;
  v_visual_revision_count smallint;
  v_satisfaction jsonb;
  v_snapshot jsonb;
  v_instruction record;
  v_context_changed boolean := false;
  v_updated integer;
begin
  select s.profile_id, pr.status, pr.published_at, pr.history_summary,
         pr.visual_qa_status, pr.visual_revision_count,
         pr.final_temporary_satisfaction, pr.context_snapshot
  into v_profile_id, v_status, v_published_at, v_history_summary,
       v_visual_qa_status, v_visual_revision_count, v_satisfaction, v_snapshot
  from public.posting_runs pr
  join public.schedules s on s.id = pr.schedule_id
  where pr.id = p_posting_run_id
  for update of pr;

  if not found or v_profile_id is null then
    raise exception 'Posting run or owned schedule was not found';
  end if;
  if v_status <> 'published' or v_published_at is null then
    raise exception 'Posting run is not durably published';
  end if;
  if v_visual_qa_status not in ('accepted', 'accepted_with_minor_issues') then
    raise exception 'Posting run does not have accepted visual QA';
  end if;
  if v_history_summary is null or btrim(v_history_summary) = '' then
    raise exception 'Posting run history summary is missing';
  end if;
  if v_satisfaction is null
     or (v_satisfaction ->> 'visualRevision')::smallint <> v_visual_revision_count then
    raise exception 'Final temporary satisfaction is missing or stale';
  end if;

  if exists (
    select 1 from public.social_media_past_posts
    where posting_run_id = p_posting_run_id
  ) then
    return false;
  end if;

  perform 1 from public.social_media_context_states
  where profile_id = v_profile_id for update;
  if not found then
    raise exception 'Context state was not found for published profile';
  end if;

  for v_instruction in
    select t.id, t.remaining_posts
    from public.social_media_temporary_instructions t
    join (
      select distinct value::uuid as id
      from jsonb_array_elements_text(v_satisfaction -> 'instructionIds')
    ) requested on requested.id = t.id
    where t.profile_id = v_profile_id
      and t.instruction_type = 'counter'
      and t.remaining_posts > 0
      and exists (
        select 1
        from jsonb_array_elements(
          coalesce(v_snapshot -> 'activeTemporaryInstructions', '[]'::jsonb)
        ) active
        where active ->> 'id' = t.id::text
      )
    order by t.id
    for update of t
  loop
    if v_instruction.remaining_posts = 1 then
      delete from public.social_media_temporary_instructions
      where id = v_instruction.id and profile_id = v_profile_id
        and instruction_type = 'counter' and remaining_posts = 1;
    else
      update public.social_media_temporary_instructions
      set remaining_posts = remaining_posts - 1, updated_at = now()
      where id = v_instruction.id and profile_id = v_profile_id
        and instruction_type = 'counter'
        and remaining_posts = v_instruction.remaining_posts;
    end if;
    get diagnostics v_updated = row_count;
    if v_updated = 1 then v_context_changed := true; end if;
  end loop;

  if v_context_changed then
    update public.social_media_context_states
    set revision = revision + 1, updated_at = now()
    where profile_id = v_profile_id;
    get diagnostics v_updated = row_count;
    if v_updated <> 1 then
      raise exception 'Context state was not found for published profile';
    end if;
  end if;

  insert into public.social_media_past_posts
    (profile_id, posting_run_id, published_at, summary)
  values (v_profile_id, p_posting_run_id, v_published_at, v_history_summary);

  return true;
end
$function$

CREATE OR REPLACE FUNCTION public.newsletter_campaign_sending_context(p_campaign_ids uuid[])
 RETURNS TABLE(campaign_id uuid, from_email text, reply_to_email text, tenant_name text, configuration_set_name text)
 LANGUAGE sql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
  select
    campaign.id,
    identity.from_email,
    identity.reply_to_email,
    tenant.tenant_name,
    tenant.configuration_set_name
  from public.newsletter_campaigns campaign
  join public.newsletter_sending_identities identity
    on identity.profile_id = campaign.profile_id
   and (campaign.newsletter_sending_identity_id is null
     or campaign.newsletter_sending_identity_id = identity.id)
  join public.newsletter_ses_tenants tenant
    on tenant.profile_id = campaign.profile_id
   and (campaign.newsletter_ses_tenant_id is null
     or campaign.newsletter_ses_tenant_id = tenant.id)
  where campaign.id = any(p_campaign_ids)
    and identity.status = 'ready'
    and identity.verification_status = 'SUCCESS'
    and identity.reply_to_verified_at is not null
    and tenant.status = 'ready';

CREATE OR REPLACE FUNCTION public.newsletter_claim_sends(p_items jsonb, p_claim_token uuid, p_lease_seconds integer DEFAULT 300)
 RETURNS TABLE(disposition text, send_id uuid, campaign_id uuid, email_address text, subject_line text, preview_text text, from_name text, from_email text, email_content jsonb, unsubscribe_token uuid, attempt_count integer)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  v_item jsonb;
  v_send public.newsletter_sends%rowtype;
  v_campaign record;
  v_contact_status text;
begin
  if jsonb_typeof(p_items) <> 'array' then
    raise exception 'p_items must be a JSON array';
  end if;
  if p_lease_seconds < 30 or p_lease_seconds > 900 then
    raise exception 'lease seconds out of range';
  end if;

  -- Deterministic lock ordering avoids deadlocks when duplicate SQS deliveries
  -- put overlapping sends into two concurrent Lambda batches.
  for v_item in
    select value
    from jsonb_array_elements(p_items)
    order by value->>'campaignId', value->>'newsletterSendId'
  loop
    disposition := null;
    send_id := (v_item->>'newsletterSendId')::uuid;
    campaign_id := (v_item->>'campaignId')::uuid;
    email_address := null;
    subject_line := null;
    preview_text := null;
    from_name := null;
    from_email := null;
    email_content := null;
    unsubscribe_token := null;
    attempt_count := null;

    select * into v_send
    from public.newsletter_sends s
    where s.id = send_id
    for update;

    if not found then
      disposition := 'not_found';
      return next;
      continue;
    end if;
    if v_send.newsletter_campaign_id is distinct from campaign_id then
      disposition := 'campaign_mismatch';
      return next;
      continue;
    end if;
    if v_send.status in ('sent', 'failed', 'cancelled', 'suppressed', 'uncertain') then
      disposition := 'terminal';
      return next;
      continue;
    end if;

    select c.*, e.content as email_content
      into v_campaign
    from public.newsletter_campaigns c
    join public.newsletter_emails e on e.id = c.newsletter_email_id
    where c.id = campaign_id;

    if not found then
      disposition := 'not_found';
      return next;
      continue;
    end if;

    if v_campaign.status in ('cancelled', 'failed') then
      update public.newsletter_sends
      set status = 'cancelled', failed_at = now(), last_error = 'Campaign cancelled before send'
      where id = send_id;
      update public.newsletter_campaigns
      set terminal_send_count = terminal_send_count + 1,
          cancelled_count = cancelled_count + 1
      where id = campaign_id;
      disposition := 'terminal';
      return next;
      continue;
    end if;

    select c.email_marketing_status into v_contact_status
    from public.crm_contacts c
    where c.id = v_send.crm_contact_id
      and c.profile_id = v_campaign.profile_id;

    if v_contact_status is distinct from 'subscribed' then
      update public.newsletter_sends
      set status = 'suppressed', failed_at = now(), last_error = 'Recipient is not eligible for marketing email'
      where id = send_id;
      update public.newsletter_campaigns
      set terminal_send_count = terminal_send_count + 1,
          suppressed_count = suppressed_count + 1
      where id = campaign_id;
      update public.newsletter_campaigns
      set status = 'completed', completed_at = coalesce(completed_at, now())
      where id = campaign_id
        and status <> 'cancelled'
        and total_send_count > 0
        and terminal_send_count >= total_send_count;
      disposition := 'terminal';
      return next;
      continue;
    end if;

    if v_send.status = 'sending' and v_send.processing_lease_expires_at > now() then
      disposition := 'busy';
      return next;
      continue;
    end if;

    if v_send.status = 'sending'
       and v_send.processing_lease_expires_at <= now()
       and v_send.ses_request_started_at is not null then
      update public.newsletter_sends
      set status = 'uncertain',
          failed_at = now(),
          last_error = 'SES request may have been accepted before the worker stopped; manual review required',
          processing_claim_token = null,
          processing_lease_expires_at = null
      where id = send_id;
      update public.newsletter_campaigns
      set terminal_send_count = terminal_send_count + 1,
          uncertain_count = uncertain_count + 1
      where id = campaign_id;
      update public.newsletter_campaigns
      set status = 'completed', completed_at = coalesce(completed_at, now())
      where id = campaign_id
        and status <> 'cancelled'
        and total_send_count > 0
        and terminal_send_count >= total_send_count;
      disposition := 'terminal';
      return next;
      continue;
    end if;

    update public.newsletter_sends
    set status = 'sending',
        processing_started_at = now(),
        processing_lease_expires_at = now() + make_interval(secs => p_lease_seconds),
        processing_claim_token = p_claim_token,
        ses_request_started_at = null,
        last_attempt_at = now(),
        attempt_count = newsletter_sends.attempt_count + 1,
        last_error = null
    where id = send_id
    returning newsletter_sends.* into v_send;

    update public.newsletter_campaigns
    set status = case when status in ('queued', 'dispatching') then 'sending' else status end
    where id = campaign_id;

    disposition := 'claimed';
    email_address := v_send.email_address;
    subject_line := v_campaign.subject_line;
    preview_text := v_campaign.preview_text;
    from_name := v_campaign.from_name;
    from_email := v_campaign.from_email;
    email_content := v_campaign.email_content;
    unsubscribe_token := v_send.unsubscribe_token;
    attempt_count := v_send.attempt_count;
    return next;
  end loop;
end;
$function$


CREATE OR REPLACE FUNCTION public.newsletter_complete_sends(p_results jsonb, p_claim_token uuid)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  v_result jsonb;
  v_send public.newsletter_sends%rowtype;
  v_outcome text;
  v_error text;
begin
  if jsonb_typeof(p_results) <> 'array' then
    raise exception 'p_results must be a JSON array';
  end if;

  for v_result in select value from jsonb_array_elements(p_results)
  loop
    select * into v_send
    from public.newsletter_sends
    where id = (v_result->>'newsletterSendId')::uuid
      and status = 'sending'
      and processing_claim_token = p_claim_token
    for update;

    if not found then
      continue;
    end if;

    v_outcome := v_result->>'outcome';
    v_error := left(coalesce(v_result->>'error', ''), 2000);

    if v_outcome = 'sent' then
      update public.newsletter_sends
      set status = 'sent',
          sent_at = now(),
          ses_message_id = left(v_result->>'sesMessageId', 255),
          last_error = null,
          processing_claim_token = null,
          processing_lease_expires_at = null
      where id = v_send.id;
      update public.newsletter_campaigns
      set terminal_send_count = terminal_send_count + 1,
          sent_count = sent_count + 1
      where id = v_send.newsletter_campaign_id;
    elsif v_outcome = 'retry' then
      update public.newsletter_sends
      set status = 'pending',
          last_error = v_error,
          ses_request_started_at = null,
          processing_claim_token = null,
          processing_lease_expires_at = null
      where id = v_send.id;
    elsif v_outcome = 'uncertain' then
      update public.newsletter_sends
      set status = 'uncertain',
          failed_at = now(),
          last_error = v_error,
          processing_claim_token = null,
          processing_lease_expires_at = null
      where id = v_send.id;
      update public.newsletter_campaigns
      set terminal_send_count = terminal_send_count + 1,
          uncertain_count = uncertain_count + 1
      where id = v_send.newsletter_campaign_id;
    else
      update public.newsletter_sends
      set status = 'failed',
          failed_at = now(),
          last_error = v_error,
          ses_request_started_at = null,
          processing_claim_token = null,
          processing_lease_expires_at = null
      where id = v_send.id;
      update public.newsletter_campaigns
      set terminal_send_count = terminal_send_count + 1,
          failed_count = failed_count + 1
      where id = v_send.newsletter_campaign_id;
    end if;

    update public.newsletter_campaigns
    set status = 'completed', completed_at = coalesce(completed_at, now())
    where id = v_send.newsletter_campaign_id
      and status <> 'cancelled'
      and total_send_count > 0
      and terminal_send_count >= total_send_count;
  end loop;
end;
$function$

CREATE OR REPLACE FUNCTION public.newsletter_prepare_campaign(p_campaign_id uuid)
 RETURNS TABLE(prepared boolean, send_count integer)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  v_campaign public.newsletter_campaigns%rowtype;
  v_count integer;
begin
  perform pg_advisory_xact_lock(hashtextextended(p_campaign_id::text, 0));

  select * into v_campaign
  from public.newsletter_campaigns
  where id = p_campaign_id
  for update;

  if not found then
    raise exception 'newsletter campaign not found';
  end if;
  if v_campaign.status = 'cancelled' then
    raise exception 'newsletter campaign is cancelled';
  end if;
  if v_campaign.recipient_preparation_status = 'prepared' then
    return query select true, v_campaign.total_send_count;
    return;
  end if;

  update public.newsletter_campaigns
  set recipient_preparation_status = 'preparing',
      recipient_preparation_started_at = coalesce(recipient_preparation_started_at, now()),
      status = 'preparing',
      last_error = null
  where id = p_campaign_id;

  insert into public.newsletter_sends (
    newsletter_campaign_id,
    crm_contact_id,
    email_address,
    status
  )
  select distinct on (lower(btrim(c.email)))
    p_campaign_id,
    c.id,
    btrim(c.email),
    'pending'
  from public.newsletter_campaign_audience_connections cac
  join public.newsletter_audiences a
    on a.id = cac.newsletter_audience_id
   and a.profile_id = v_campaign.profile_id
  join public.newsletter_audience_contact_connections acc
    on acc.newsletter_audience_id = a.id
  join public.crm_contacts c
    on c.id = acc.crm_contact_id
   and c.profile_id = v_campaign.profile_id
  where cac.newsletter_campaign_id = p_campaign_id
    and c.email is not null
    and btrim(c.email) <> ''
    and c.email_marketing_status = 'subscribed'
  order by lower(btrim(c.email)), c.id
  on conflict do nothing;

  select count(*)::integer into v_count
  from public.newsletter_sends
  where newsletter_campaign_id = p_campaign_id;

  update public.newsletter_campaigns
  set recipient_preparation_status = 'prepared',
      recipient_prepared_at = now(),
      total_send_count = v_count,
      status = case when v_count = 0 then 'completed' else 'dispatching' end,
      completed_at = case when v_count = 0 then now() else completed_at end
  where id = p_campaign_id;

  return query select true, v_count;
end;
$function$
